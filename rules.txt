rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // --- HELPER FUNCTIONS ---
    function isSignedIn() {
      return request.auth != null;
    }
    
    function getProfile() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    
    function hasProfile() {
      return exists(/databases/$(database)/documents/users/$(request.auth.uid));
    }
    
    function isSuperAdmin() {
      return isSignedIn() && hasProfile() && getProfile().get('role', 'user') == 'super_admin';
    }
    
    function isAdmin() {
      return isSignedIn() && hasProfile() && getProfile().get('role', 'user') == 'admin';
    }

    // --- USER PROFILES ---
    match /users/{userId} {
      allow read: if isSignedIn() && (request.auth.uid == userId || isSuperAdmin() || isAdmin());
      
      allow create: if isSignedIn() &&
        request.auth.uid == userId &&
        request.resource.data.get('role', 'user') == 'user' &&
        request.resource.data.get('accessible_banks', {}) == {};
        
      allow update: if isSuperAdmin() || (
        isSignedIn() && 
        request.auth.uid == userId &&
        !request.resource.data.diff(resource.data).affectedKeys()
          .hasAny(['role', 'is_disabled', 'accessible_banks', 'can_view_inbox', 'can_message', 'can_access_quizzes'])
      );

      allow delete: if isSuperAdmin();
    }

    // --- QUESTION BANKS ---
    match /question_banks/{bankId} {
      allow read: if isSuperAdmin() || (
        isSignedIn() &&
        hasProfile() && 
        getProfile().get('accessible_banks', {})[bankId] == true
      ) || (
        resource != null && 
        resource.data.get('created_by', '') == request.auth.uid
      );
      
      allow create: if isSuperAdmin() || isAdmin();
      
      allow update, delete: if isSuperAdmin() || isAdmin() || (
        resource != null && 
        resource.data.get('created_by', '') == request.auth.uid
      );

      match /cloud_questions/{qId} {
        allow read: if isSuperAdmin() || isAdmin() || (
          isSignedIn() &&
          hasProfile() && 
          getProfile().get('accessible_banks', {})[bankId] == true
        ) || 
        get(/databases/$(database)/documents/question_banks/$(bankId)).data.get('created_by', '') == request.auth.uid;
          
        allow write: if isSuperAdmin() || isAdmin() ||
        get(/databases/$(database)/documents/question_banks/$(bankId)).data.get('created_by', '') == request.auth.uid;
      }
    }

    // --- APP CONFIG ---
    match /app_config/{configId} {
      allow read: if isSignedIn();
      allow write: if isSuperAdmin();
    }

    // --- MESSAGES ---
    match /messages/{messageId} {
      // Allow creation if signed in and NOT specifically banned (if profile exists)
      allow create: if isSignedIn() && 
        request.resource.data.get('sender_uid', '') == request.auth.uid &&
        (!hasProfile() || getProfile().get('can_message', true) == true);
      
      // Allow read if sender or admin-with-inbox-access
      allow read: if (
        isSignedIn() && 
        resource.data.get('sender_uid', '') == request.auth.uid
      ) || (
        isSuperAdmin() || 
        isAdmin() || 
        (hasProfile() && getProfile().get('can_view_inbox', false) == true)
      );

      allow update: if isSuperAdmin() || isAdmin();
      allow delete: if isSuperAdmin();
    }
  }
}
